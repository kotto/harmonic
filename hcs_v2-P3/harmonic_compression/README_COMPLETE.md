### **Usage Simple**
```python
from harmonic_compression.core import harmonic_engine
import numpy as np

# Image de test
image = np.random.randint(0, 256, (200, 300, 3), dtype=np.uint8)

# Compression automatique avec optimisation
result = harmonic_engine.compress_image(image, energy_level='standard')

if result['success']:
    print(f"✅ Compression: {result['compression_ratio']:.1f}:1")
    print(f"🌊 Mode: {result['mode_used']}")
    print(f"⏱️ Temps: {result['processing_time']:.3f}s")
```
