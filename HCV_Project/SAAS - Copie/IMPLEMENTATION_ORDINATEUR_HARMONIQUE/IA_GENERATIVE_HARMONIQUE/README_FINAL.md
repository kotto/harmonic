## 🎯 Utilisation Rapide

### 🧠 Réseau Neuronal Harmonique

```python
from harmonic_neural_network import HarmonicNeuralNetwork, ActivationType, OptimizationType

# Création du réseau
network = HarmonicNeuralNetwork(
    layers=[1024, 512, 256, 128],
    activation=ActivationType.HARMONIC_SIGMOID,
    optimization=OptimizationType.PHI_ADAM
)

# Entraînement
import numpy as np
X = np.random.randn(1000, 1024)
y = np.random.randn(1000, 128)

results = network.train_harmonic(X, y, epochs=50, learning_rate=0.01)

print(f"🧠 Entraînement terminé:")
print(f"   Temps: {results['training_time']:.2f}s")
print(f"   Score harmonique: {results['metrics']['harmonic_score']:.3f}")
```

### 🚀 Génération de Code Harmonique

```python
from harmonic_code_generator import HarmonicCodeGenerator, CodeRequirements

# Création du générateur
generator = HarmonicCodeGenerator()

# Configuration des requirements
requirements = CodeRequirements(
    service_name="QuantiqueHarmonique",
    description="Service de calcul quantique harmonique",
    endpoints=[
        {"method": "POST", "path": "/factorization", "description": "Factorisation harmonique"}
    ],
    database_schema={"table": "quantique_jobs", "fields": ["id", "name", "result", "status"]},
    business_logic=["Optimisation φ des calculs", "Précision π des résultats"],
    validation_rules=["Nom requis", "Valeur numérique"]
)

# Génération du code
generated_files = generator.generate_full_application(requirements, "./output")

print(f"🚀 Code généré: {len(generated_files)} fichiers")
for file_type, file_path in generated_files.items():
    print(f"   {file_type}: {file_path}")
```

### ⚡ Optimisation Harmonique

```python
from harmonic_optimization_engine import HarmonicOptimizationEngine, OptimizationTarget, OptimizationType

# Configuration du moteur
engine = HarmonicOptimizationEngine()

# Définition des cibles
targets = [
    OptimizationTarget(
        name="performance_phi",
        current_value=1.0,
        target_value=1.618,  # φ
        optimization_type=OptimizationType.PHI_PERFORMANCE
    ),
    OptimizationTarget(
        name="precision_pi",
        current_value=3.0,
        target_value=3.141593,  # π
        optimization_type=OptimizationType.PI_PRECISION
    ),
    OptimizationTarget(
        name="efficiency_e",
        current_value=2.5,
        target_value=2.718282,  # e
        optimization_type=OptimizationType.E_EFFICIENCY
    )
]

# Optimisation
result = engine.optimize_targets(targets)

print(f"⚡ Optimisation terminée:")
print(f"   Amélioration: {result.improvement:.2f}%")
print(f"   Score harmonique: {result.harmonic_score:.3f}")
print(f"   Itérations: {result.iterations}")
```

### 🧠 Système de Mémoire Harmonique

```python
from harmonic_memory_system import HarmonicMemorySystem, MemoryConfig, MemoryType, MemoryPriority

# Configuration
config = MemoryConfig(max_entries=10000, cache_size=1000)
memory = HarmonicMemorySystem(config)

# Stockage
entry_id = memory.store(
    content="Calcul quantique φ-optimisé",
    memory_type=MemoryType.PROCEDURAL,
    priority=MemoryPriority.HIGH,
    tags=["quantique", "phi", "optimization"]
)

# Récupération
results = memory.retrieve("quantique", limit=5)

print(f"🧠 Mémoire:")
print(f"   Entrées stockées: {len(entry_id)}")
print(f"   Résultats récupérés: {len(results)}")
```
