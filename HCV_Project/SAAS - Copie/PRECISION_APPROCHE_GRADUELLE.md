def optimisation_graduelle(constante_name, formule_actuelle, precision_actuelle):
    """
    Optimisation progressive d'une constante harmonique
    """
    
    # Étape 1 : Analyse de la structure actuelle
    structure = analyser_structure(formule_actuelle)
    
    # Étape 2 : Génération de variantes
    variantes = generer_variantes(structure)
    
    # Étape 3 : Test systématique
    resultats = []
    for variante in variantes:
        precision = calculer_precision(variante)
        if precision > precision_actuelle:
            resultats.append({
                'formule': variante,
                'precision': precision,
                'gain': precision - precision_actuelle
            })
    
    # Étape 4 : Sélection des meilleures
    resultats.sort(key=lambda x: x['precision'], reverse=True)
    
    return resultats[:5]  # Top 5 des améliorations
